% \colocar{t|c|s}{contenido}: coloca el contenido arriba (t), centrado en vertical (c)
% o repartido en todo el alto con \vfill (s); lo reduce si no cabe y dibuja el pie
% con la fuente de los datos y el número de página.
\newsavebox\cajadiapo
\newlength\altocaja
\newcommand\colocar[2]{%
  \if s#1\relax
    \sbox\cajadiapo{\begin{minipage}[t][\altodisponible][s]{\anchodisponible}#2\end{minipage}}%
  \else
    \sbox\cajadiapo{\begin{minipage}[t]{\anchodisponible}#2\end{minipage}}%
  \fi
  \setlength\altocaja{\dimexpr\ht\cajadiapo+\dp\cajadiapo\relax}%
  \ifdim\altocaja>\dimexpr\altodisponible+0.5pt\relax
    \typeout{AJUSTE: diapositiva \arabic{framenumber} reducida (\the\altocaja)}%
    \sbox\cajadiapo{\resizebox*{!}{\altodisponible}{\usebox\cajadiapo}}%
  \fi
  \begin{tikzpicture}[remember picture,overlay]
    \if c#1\relax
      \node[anchor=west,inner sep=0pt] at ([xshift=30.35pt,yshift=-124.24pt]current page.north west) {\usebox\cajadiapo};
    \else
      \node[anchor=north west,inner sep=0pt] at ([xshift=30.35pt,yshift=-30.35pt]current page.north west) {\usebox\cajadiapo};
    \fi
    % La letra va en las opciones del nodo: así align=left mide los espacios con ella
    \node[anchor=south west,inner sep=0pt,text width=\anchodisponible,align=left,
          font=\fontsize{5.69}{7.1}\selectfont,text=\colorpie]
      at ([xshift=30.35pt,yshift=15.17pt]current page.south west) {\piediapo};
    \node[anchor=south east,inner sep=0pt,font=\fontsize{5.69}{7.1}\selectfont,text=\colorpie]
      at ([xshift=-10pt,yshift=15.17pt]current page.south east) {\insertframenumber};
  \end{tikzpicture}}
